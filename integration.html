async function init(){
  const user=await requireUser();if(!user)return;wireLogout();
  if(siteId){
    const r=await db.from("sites").select("id,name,site_key,status,allowed_domains").eq("id",siteId).eq("owner_id",user.id).maybeSingle();
    if(!r.data){siteId="";siteKey="YOUR_SITE_KEY";}
    else{siteKey=r.data.site_key;siteName=r.data.name;siteStatus=r.data.status;siteAllowedDomains=r.data.allowed_domains||[];$("#backSite").href="website.html?id="+encodeURIComponent(r.data.id);$("#integrationSiteName").textContent=r.data.name;}
  }else{siteKey=params.get("site_key")||"YOUR_SITE_KEY";}
  renderSnippets();
  const firstDomain=siteAllowedDomains.find(value=>!String(value||"").trim().startsWith("*."));
  if(firstDomain)$("#siteUrl").value="https://"+normalizeDomain(firstDomain);
  $("#siteUrl").addEventListener("input",()=>{const value=$("#siteUrl").value.trim();if(!value){$("#siteUrlStatus").textContent="Use a domain already registered in Website Settings.";return;}try{const parsed=new URL(value);const ok=isAllowedWebsiteHost(parsed.hostname);$("#siteUrlStatus").textContent=ok?"Domain confirmed against this site’s Allowed Domains list.":"This hostname is not allowed yet. Add it in Website Settings first.";$("#siteUrlStatus").style.color=ok?"":"#a53232";}catch{$("#siteUrlStatus").textContent="Enter a complete URL such as https://example.com";$("#siteUrlStatus").style.color="#a53232";}});
  $("#generatePackage").addEventListener("click",generatePackage);
  renderDemo();
}
function renderDemo(){
  const labLink=$("#botLabLink");if(labLink)labLink.href=siteId?"bot-lab.html?site_id="+encodeURIComponent(siteId):"bot-lab.html";
  const demo=$("#widgetDemo");if(!demo)return;demo.replaceChildren();
  const title=document.createElement("h3");title.textContent=siteId?"Live CAPTCHA test":"Widget preview";
  const description=document.createElement("p");description.className="desc";
  demo.append(title,description);
  if(!siteId){
    description.textContent="Open this page from a website you own in Dashboard to load its real CAPTCHA. Code examples use a placeholder Site Key until a website is selected.";
    const preview=document.createElement("div");preview.className="integration-demo";preview.innerHTML="Human verification✓ BeeCaptcha widget areaSelect an owned website to run a real challenge.";demo.appendChild(preview);return;
  }
  if(siteStatus!=="active"){description.textContent="This site is disabled. Enable it in Website Settings before running the live test.";return;}
  if(!allowedOnCurrentHost()){
    description.textContent="The live challenge is not loaded because this dashboard's domain is not in the site's allowed-domain list. Test the embed on one of the domains you registered; domain protection has not been bypassed.";
    const note=document.createElement("div");note.className="integration-warning";note.textContent="Allowed domains: "+(siteAllowedDomains.join(", ")||"none");demo.appendChild(note);return;
  }
  description.textContent="This runs a real challenge and tests verification in the dashboard. Production websites must still validate tokens with captcha-verify and their server-only Secret Key.";
  const widget=document.createElement("vin-captcha");widget.dataset.vinCaptcha=siteKey;widget.className="integration-live-widget";
  const status=document.createElement("p");status.className="integration-live-status";status.textContent="Loading the live widget…";
  widget.addEventListener("vin-captcha-start",()=>{status.textContent="Loading a fresh CAPTCHA challenge…";});
  widget.addEventListener("vin-captcha-challenge",()=>{status.textContent="New challenge loaded. Complete it to test server-side verification.";});
  widget.addEventListener("vin-captcha-error",(event)=>{status.textContent="CAPTCHA could not load: "+String(event.detail?.code||"unknown")+". Try refreshing the challenge.";});
  widget.addEventListener("vin-captcha-verified",async(event)=>{
    status.textContent="Checking the answer on the BeeCaptcha server…";
    try{
      const sessionResult=await db.auth.getSession();const accessToken=sessionResult.data?.session?.access_token;
      if(sessionResult.error||!accessToken){status.textContent="Sign in again to run the server verification test.";return;}
      const response=await fetch(SUPABASE_URL+"/functions/v1/captcha-test-verify",{method:"POST",headers:{"content-type":"application/json","Authorization":"Bearer "+accessToken,"apikey":SUPABASE_PUBLISHABLE_KEY},body:JSON.stringify({site_id:siteId,verification_token:event.detail.token})});
      let result={};try{result=await response.json();}catch{}
      if(response.ok&&result.success===true){status.textContent="✓ Server verified the answer, challenge token, expiry, and single-use status. This dashboard test does not expose your Secret Key.";}
      else{status.textContent="✕ Server rejected the challenge: "+String(result.code||"verification_failed")+". Refresh the page to start a fresh challenge.";}
    }catch{status.textContent="Could not reach the server verification test. Check the network and refresh the page.";}
  });
  demo.append(widget,status);
  const script=document.createElement("script");script.src=WIDGET_URL+"?site_key="+encodeURIComponent(siteKey);script.async=true;
  script.onerror=()=>{status.textContent="Could not load the widget script. Check the function deployment and reload the page.";};
  demo.appendChild(script);
}
